177class TaskQueue {
178
179private:
180 using WrapperFunc = std::function<void()>;
181 std::deque<WrapperFunc> queue_{};
182 std::thread thread_{};
183 std::mutex mut_{};
184 std::mutex qMut_{};
185 std::condition_variable cv_{};
186 std::atomic<bool> running_{false};
187 std::atomic<bool> repeatStop_{false};
188
189 void run_()
190 {
191 while(running_.load()) {
192 std::unique_lock cvLock(mut_);
193 cv_.wait(cvLock, [&]() -> bool {
194 std::lock_guard lock(qMut_);
195 return !queue_.empty() || !running_.load();
196 });
197
198 if (!running_.load()) {
199 break;
200 }
201
202 WrapperFunc func;
203 {
204 std::lock_guard lock(qMut_);
205 func = std::move(queue_.front());
206 queue_.pop_front();
207 if (!queue_.empty()) {
208 cv_.notify_one();
209 }
210 }
211 // The task can take a long time and we want to minimize mutex holding time.
212 // So we relase the mutex after we are done with the queue.
213 func();
214 }
215 }
216
217 template<typename Callable, typename... Args>
218 void postRepeat_(Callable&& callable, Args&&... args)
219 {
220 std::function<decltype(callable(args...))()> func =
221 std::bind(std::forward<Callable>(callable), std::forward<Args>(args)...);
222 std::lock_guard lock(qMut_);
223 queue_.emplace_back(WrapperFunc([&, f=std::move(func), callable, args...]() {
224 if (repeatStop_.load()) {
225 return;
226 }
227 f();
228 postRepeat_(callable, args...);
229 }));
230 if (queue_.size() == 1) {
231 cv_.notify_one();
232 }
233 }
234
235public:
236 TaskQueue() : running_(true), repeatStop_(true)
237 {
238 thread_ = std::thread(std::bind(&TaskQueue::run_, this));
239 }
240
241 ~TaskQueue()
242 {
243 running_ = false;
244 cv_.notify_all();
245 if (thread_.joinable()) {
246 thread_.join();
247 }
248 }

250 template<typename Callable, typename... Args>
251 auto enqueue(Callable&& callable, Args&&... args) -> std::future<decltype(callable(args...))>
252 {
253 std::function<decltype(callable(args...))()> func =
254 std::bind(std::forward<Callable>(callable), std::forward<Args>(args)...);
255 auto task = std::make_shared<std::packaged_task<decltype(callable(args...))()>>(func);
256 std::lock_guard lock(qMut_);
257 queue_.emplace_back(WrapperFunc([task]() {
258 (*task)();
259 }));
260 if (queue_.size() == 1) {
261 cv_.notify_one();
262 }
263 return task->get_future();
264 }
265
266 template<typename Callable, typename... Args>
267 void startRepeat(Callable&& callable, Args&&... args)
268 {
269 if (repeatStop_.load()) {
270 repeatStop_ = false;
271 postRepeat_(std::forward<Callable>(callable), std::forward<Args>(args)...);
272 }
273 }

275 void stopRepeat()

277 repeatStop_ = true;
